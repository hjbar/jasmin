% To print ratio
\newcommand{\printRatioA}[1]{%
  \edef\colorResult{\fpeval{#1 < 1.0 ? 1 : 0}}%
  \edef\underlineResult{\fpeval{(#1 <= 0.95) || (#1 >= 1.05) ? 1 : 0}}%
  \ifnum\colorResult=1 %
    \def\coloredText{{\color{red}#1}}%
  \else%
    \def\coloredText{{\color{green!60!black}#1}}%
  \fi%
  \ifnum\underlineResult=1 %
    \uline{\coloredText}%
  \else%
    \coloredText%
  \fi%
}
